10const NUMBER_OF_DIRTY_AREAS: usize = 64;
11const DIRTY_AREA_SIZE: usize = (1 << 16) / NUMBER_OF_DIRTY_AREAS;
12
13// Storage is allocated at a finer granularity than the dirty-area granularity:
14// each sub-chunk holds `SUBCHUNK_SLOTS` slots, so a frame that scatters writes
15// allocates only the sub-chunks it actually touches, not a full 32 KiB area.
16// This bounds the "scatter one write into every area to force 2 MiB/frame"
17// pattern (deep nested calls) and, because far-call frames no longer pay a
18// large per-frame zeroing, it also *reduces* cycles vs. both the dense stack
19// and the coarser chunking. Dirty tracking stays at the coarse area level (the
20// `dirty_areas` u64), so snapshot/rollback/equality semantics are unchanged.
21const SUBCHUNK_SLOTS: usize = 16;
22const NUM_SUBCHUNKS: usize = (1 << 16) / SUBCHUNK_SLOTS;
23const SUBCHUNKS_PER_AREA: usize = DIRTY_AREA_SIZE / SUBCHUNK_SLOTS;
24
25/// A contiguous block of `SUBCHUNK_SLOTS` stack slots, allocated on demand.
26type SlotChunk = Box<[U256; SUBCHUNK_SLOTS]>;
27
28/// Allocate a zeroed slot chunk without materializing it on the caller's stack
29/// frame first. `U256`'s all-zero bit pattern is the integer zero, so
30/// `alloc_zeroed` is sound.
31#[allow(clippy::cast_ptr_alignment)] // aligned per the array layout
32fn zeroed_chunk() -> SlotChunk {
33    let layout = Layout::new::<[U256; SUBCHUNK_SLOTS]>();
34    // `alloc_zeroed` returns null on failure; wrapping null in a `Box` is UB, so bail
35    // out through `handle_alloc_error` (a clean abort) before `Box::from_raw`.
36    unsafe {
37        let ptr = alloc_zeroed(layout);
38        if ptr.is_null() {
39            handle_alloc_error(layout);
40        }
41        Box::from_raw(ptr.cast())
42    }
43}

45/// VM stack.
46///
47/// The slots are stored as [`NUM_SUBCHUNKS`] sub-chunks that are allocated
48/// lazily on first write, so a frame that touches only a handful of slots pays
49/// for a couple of small sub-chunks rather than the full 2 MiB. An absent
50/// sub-chunk reads as all-zero, exactly like the dense zero-initialized array
51/// it replaces — this is purely a memory-layout change with no observable
52/// difference. Keeping the backing sparse bounds the memory held by deep call
53/// stacks (each live frame keeps its own `Stack`, and `StackPool` retains them
54/// for reuse).
55#[derive(Clone)]
56pub(crate) struct Stack {
57    /// set of slots that may be interpreted as [`FatPointer`].
58    pointer_flags: Bitset,
59    dirty_areas: u64,
60    slots: [Option<SlotChunk>; NUM_SUBCHUNKS],
61}
62
63impl Stack {
64    #[allow(clippy::cast_ptr_alignment)] // aligned per `Stack` layout
65    pub(crate) fn new() -> Box<Self> {
66        // A zeroed `Stack` is valid: `Bitset` is all-zero, `dirty_areas` is 0,
67        // and `Option<Box<_>>` uses the null-pointer niche, so all chunks are `None`.
68        let layout = Layout::new::<Self>();
69        // `alloc_zeroed` returns null on failure; wrapping null in a `Box` is UB, so bail
70        // out through `handle_alloc_error` (a clean abort) before `Box::from_raw`.
71        unsafe {
72            let ptr = alloc_zeroed(layout);
73            if ptr.is_null() {
74                handle_alloc_error(layout);
75            }
76            Box::from_raw(ptr.cast())
77        }
78    }
79
80    #[inline(always)]
81    pub(crate) fn get(&self, slot: u16) -> U256 {
82        let subchunk = slot as usize / SUBCHUNK_SLOTS;
83        match &self.slots[subchunk] {
84            Some(chunk) => chunk[slot as usize % SUBCHUNK_SLOTS],
85            None => U256::zero(),
86        }
87    }
88
89    #[inline(always)]
90    pub(crate) fn set(&mut self, slot: u16, value: U256) {
91        let area = slot as usize / DIRTY_AREA_SIZE;
92        // Mark the (coarse) area dirty on every write (including zero writes),
93        // matching the previous implementation so `dirty_areas` evolves
94        // identically and snapshot/rollback/eq are unaffected.
95        self.dirty_areas |= 1 << area;
96        let subchunk = slot as usize / SUBCHUNK_SLOTS;
97        let chunk = self.slots[subchunk].get_or_insert_with(zeroed_chunk);
98        chunk[slot as usize % SUBCHUNK_SLOTS] = value;
99    }
100
101    fn zero(&mut self) {
102        // Dropping a sub-chunk returns it to all-zero (absent reads as zero).
103        // A sub-chunk can only be allocated within a dirty area, so clearing
104        // every dirty area's sub-chunks clears everything.
105        for i in 0..NUMBER_OF_DIRTY_AREAS {
106            if self.dirty_areas & (1 << i) != 0 {
107                for sc in (i * SUBCHUNKS_PER_AREA)..((i + 1) * SUBCHUNKS_PER_AREA) {
108                    self.slots[sc] = None;
109                }
110            }
111        }
112
113        self.dirty_areas = 0;
114        self.pointer_flags = Bitset::default();
115    }
116
117    #[inline(always)]
118    pub(crate) fn get_pointer_flag(&self, slot: u16) -> bool {
119        self.pointer_flags.get(slot)
120    }
121
122    #[inline(always)]
123    pub(crate) fn set_pointer_flag(&mut self, slot: u16) {
124        self.pointer_flags.set(slot);
125    }
126
127    #[inline(always)]
128    pub(crate) fn clear_pointer_flag(&mut self, slot: u16) {
129        self.pointer_flags.clear(slot);
130    }

132    pub(crate) fn snapshot(&self) -> StackSnapshot {
133        let dirty_prefix_end = NUMBER_OF_DIRTY_AREAS - self.dirty_areas.leading_zeros() as usize;
134
135        // Materialize the same dense prefix the dense implementation stored:
136        // areas `0..dirty_prefix_end`, with absent sub-chunks contributing zeros.
137        let mut slots = vec![U256::zero(); DIRTY_AREA_SIZE * dirty_prefix_end];
138        for sc in 0..(dirty_prefix_end * SUBCHUNKS_PER_AREA) {
139            if let Some(chunk) = &self.slots[sc] {
140                slots[sc * SUBCHUNK_SLOTS..(sc + 1) * SUBCHUNK_SLOTS].copy_from_slice(&chunk[..]);
141            }
142        }
143
144        StackSnapshot {
145            pointer_flags: self.pointer_flags.clone(),
146            dirty_areas: self.dirty_areas,
147            slots: slots.into_boxed_slice(),
148        }
149    }
150
151    pub(crate) fn rollback(&mut self, snapshot: StackSnapshot) {
152        let StackSnapshot {
153            pointer_flags,
154            dirty_areas,
155            slots,
156        } = snapshot;
157
158        self.zero();
159
160        self.pointer_flags = pointer_flags;
161        self.dirty_areas = dirty_areas;
162        // Every dirty area lies within the saved prefix (`dirty_prefix_end` is
163        // the highest dirty area + 1), so restore its sub-chunks from the prefix.
164        // Non-dirty areas stay absent and read as zero, exactly as in the snapshot.
165        for i in 0..NUMBER_OF_DIRTY_AREAS {
166            if dirty_areas & (1 << i) != 0 {
167                for sc in (i * SUBCHUNKS_PER_AREA)..((i + 1) * SUBCHUNKS_PER_AREA) {
168                    let mut chunk = zeroed_chunk();
169                    chunk.copy_from_slice(&slots[sc * SUBCHUNK_SLOTS..(sc + 1) * SUBCHUNK_SLOTS]);
170                    self.slots[sc] = Some(chunk);
171                }
172            }
173        }
174    }
175}

177impl PartialEq for Stack {
178    fn eq(&self, other: &Self) -> bool {
179        if self.dirty_areas != other.dirty_areas || self.pointer_flags != other.pointer_flags {
180            return false;
181        }
182        // Compare slot values, treating an absent sub-chunk as all-zero. This
183        // reproduces the derived comparison over the previous dense array.
184        (0..NUM_SUBCHUNKS).all(|sc| match (&self.slots[sc], &other.slots[sc]) {
185            (Some(a), Some(b)) => a == b,
186            (Some(chunk), None) | (None, Some(chunk)) => chunk.iter().all(U256::is_zero),
187            (None, None) => true,
188        })
189    }
190}

198#[derive(Debug, Default)]
199pub(crate) struct StackPool {
200    stacks: Vec<Box<Stack>>,
201}
202
203impl StackPool {
204    pub(crate) fn get(&mut self) -> Box<Stack> {
205        self.stacks.pop().map_or_else(Stack::new, |mut s| {
206            s.zero();
207            s
208        })
209    }
210
211    pub(crate) fn recycle(&mut self, stack: Box<Stack>) {
212        self.stacks.push(stack);
213    }

218/// Helper wrapper for debugging [`Stack`] / [`StackSnapshot`] contents.
219struct StackStart<I>(I);
220
221impl<I: Iterator<Item = (bool, U256)> + Clone> fmt::Debug for StackStart<I> {
222    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
223        let mut list = formatter.debug_list();
224        for (is_pointer, slot) in self.0.clone() {
225            if is_pointer {
226                list.entry(&FatPointer::from(slot));
227            } else {
228                list.entry(&slot);
229            }
230        }
231        list.finish()
232    }
233}
234
235impl fmt::Debug for Stack {
236    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
237        const DEBUGGED_SLOTS: u16 = 256;
238
239        let slots = (0..DEBUGGED_SLOTS).map(|idx| (self.pointer_flags.get(idx), self.get(idx)));
240        let materialized: Vec<U256> = (0..DEBUGGED_SLOTS).map(|idx| self.get(idx)).collect();
241        formatter
242            .debug_struct("Stack")
243            .field("start", &StackStart(slots))
244            .field(
245                "pointer_flags.hash",
246                &hash_for_debugging(&self.pointer_flags),
247            )
248            .field("slots.hash", &hash_for_debugging(&materialized))
249            .finish_non_exhaustive()
250    }
251}
252
253impl fmt::Debug for StackSnapshot {
254    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
255        const DEBUGGED_SLOTS: usize = 256;
256
257        let slots = (0_u16..)
258            .zip(&self.slots[..])
259            .map(|(idx, slot)| (self.pointer_flags.get(idx), *slot))
260            .take(DEBUGGED_SLOTS);
261        formatter
262            .debug_struct("StackSnapshot")
263            .field("dirty_areas", &self.dirty_areas)
264            .field("start", &StackStart(slots))
265            .field(
266                "pointer_flags.hash",
267                &hash_for_debugging(&self.pointer_flags),
268            )
269            .field("slots.hash", &hash_for_debugging(&self.slots))
270            .finish_non_exhaustive()
271    }
272}

286    // --- Differential fuzz vs a dense oracle -------------------------------
287    // The chunked stack is not exercised by the `single_instruction_test`
288    // divergence harness (that feature substitutes a mock stack), so these
289    // tests fuzz the real implementation against a dense `[U256; 1 << 16]`
290    // oracle to guarantee the sparse layout is observably identical.
291
292    struct XorShift(u64);
293    impl XorShift {
294        fn next(&mut self) -> u64 {
295            let mut x = self.0;
296            x ^= x << 13;
297            x ^= x >> 7;
298            x ^= x << 17;
299            self.0 = x;
300            x
301        }
302        fn slot(&mut self) -> u16 {
303            u16::try_from(self.next() & 0xffff).expect("masked to 16 bits")
304        }
305        fn value(&mut self) -> U256 {
306            U256([self.next(), self.next(), self.next(), self.next()])
307        }
308    }
309
310    const DENSE: usize = 1 << 16;
311
312    #[test]
313    fn differential_random_set_get_matches_dense_oracle() {
314        let mut rng = XorShift(0x1234_5678_9abc_def1);
315        let mut stack = Stack::new();
316        let mut oracle = vec![U256::zero(); DENSE];
317
318        for _ in 0..20_000 {
319            let slot = rng.slot();
320            // Mix in zero writes: they must allocate/dirty exactly like nonzero.
321            let value = if rng.next().is_multiple_of(8) {
322                U256::zero()
323            } else {
324                rng.value()
325            };
326            stack.set(slot, value);
327            oracle[usize::from(slot)] = value;
328        }
329        for (slot, expected) in oracle.iter().enumerate() {
330            let slot = u16::try_from(slot).unwrap();
331            assert_eq!(stack.get(slot), *expected, "mismatch at slot {slot}");
332        }
333    }

350    #[test]
351    fn snapshot_rollback_restores_exact_state() {
352        let mut rng = XorShift(0x0abc_1234_5678_9def);
353        let mut stack = Stack::new();
354
355        for _ in 0..8_000 {
356            stack.set(rng.slot(), rng.value());
357        }
358        // Capture the exact slot state at snapshot time.
359        let mut expected = vec![U256::zero(); DENSE];
360        for (slot, e) in expected.iter_mut().enumerate() {
361            *e = stack.get(u16::try_from(slot).unwrap());
362        }
363        let snap = stack.snapshot();
364
365        // Diverge: overwrite, zero-write, and touch new areas.
366        for _ in 0..8_000 {
367            stack.set(rng.slot(), rng.value());
368        }
369        stack.set(0xffff, rng.value());
370        stack.set(0, U256::zero());
371
372        stack.rollback(snap);
373        for (slot, e) in expected.iter().enumerate() {
374            let slot = u16::try_from(slot).unwrap();
375            assert_eq!(stack.get(slot), *e, "rollback mismatch at slot {slot}");
376        }
377    }

379    #[test]
380    fn eq_independent_of_write_path() {
381        // Two stacks reaching the same logical values by different paths — one
382        // that writes then zeroes a slot (present-but-zero chunk) and one that
383        // never touches it (absent chunk) — must compare equal, because both
384        // touch the same *areas* (dirty_areas must match for eq).
385        let mut a = Stack::new();
386        let mut b = Stack::new();
387
388        // Same dirty areas: both write area 0 (slot 5) and area 3 (slot 3100).
389        a.set(5, U256::from(42u64));
390        a.set(3100, U256::from(7u64));
391        a.set(6, U256::from(99u64));
392        a.set(6, U256::zero()); // present-but-zero in a
393
394        b.set(5, U256::from(42u64));
395        b.set(3100, U256::from(7u64));
396        b.set(6, U256::zero()); // dirties area 0 without a prior nonzero
397
398        assert_eq!(&a, &b, "present-zero vs freshly-zeroed slot must be equal");
399
400        // Differing value must be unequal.
401        let mut c = b.clone();
402        c.set(5, U256::from(43u64));
403        assert_ne!(&b, &c);
404    }
